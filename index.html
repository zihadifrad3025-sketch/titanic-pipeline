Advanced Data Wrangling

Group-based imputation, outlier capping, and rich feature engineering — all in Pandas.

Why group-based imputation? Filling missing Age with the global median ignores that 1st-class passengers skew older. We group by Pclass + Sex + Title (extracted from name strings) to get context-aware estimates — a massive accuracy improvement.
feature_engineering.py 
import pandas as pd
import numpy as np

# ── Load Data ────────────────────────────────────────────────
df = pd.read_csv('train.csv')

# ── Extract Title from Name ───────────────────────────────────
df['Title'] = df['Name'].str.extract(r' ([A-Za-z]+)\.', expand=False)
rare = ['Lady','Countess','Capt','Col','Don','Dr',
        'Major','Rev','Sir','Jonkheer','Dona']
df['Title'] = df['Title'].replace(rare, 'Rare')
df['Title'] = df['Title'].replace({'Mlle':'Miss', 'Ms':'Miss',
                                     'Mme':'Mrs'})

# ── Group-based Age Imputation ────────────────────────────────
def fill_age(row, medians):
    if pd.isnull(row['Age']):
        key = (row['Pclass'], row['Sex'], row['Title'])
        return medians.get(key, df['Age'].median())
    return row['Age']

age_medians = df.groupby(['Pclass','Sex','Title'])['Age'].\
              median().to_dict()
df['Age'] = df.apply(fill_age, axis=1, medians=age_medians)

# ── Fare Outlier Capping (IQR) ────────────────────────────────
Q1 = df['Fare'].quantile(0.25)
Q3 = df['Fare'].quantile(0.75)
IQR = Q3 - Q1
upper = Q3 + 1.5 * IQR
df['Fare'] = df['Fare'].clip(upper=upper)   # cap, not drop

# ── Feature Engineering ──────────────────────────────────────
df['FamilySize'] = df['SibSp'] + df['Parch'] + 1
df['IsAlone']   = (df['FamilySize'] == 1).astype(int)
df['AgeBin']    = pd.cut(df['Age'],
                    bins=[0,12,18,35,60,100],
                    labels=['Child','Teen','Adult','MidAge','Senior'])
df['LogFare']   = np.log1p(df['Fare'])
df['FareBin']   = pd.qcut(df['LogFare'], q=4,
                    labels=['Low','Med','High','VeryHigh'])
df['Deck'] = df['Cabin'].str[0].fillna('U')  # U = Unknown

print(df[['Title','FamilySize','AgeBin','Deck']].head(10))
IQR Outlier Capping: Instead of removing outlier fares (losing data), we cap them at Q3 + 1.5×IQR. A £512 fare becomes £65 — still "high" but not a statistical monster that warps gradient descent.

# ── One-Hot Encoding (pure NumPy) ────────────────────────────
def one_hot(arr):
    """Encode a 1D array of labels into a 2D one-hot matrix."""
    classes = np.unique(arr)
    result  = np.zeros((len(arr), len(classes)), dtype=int)
    for i, cls in enumerate(classes):
        result[:, i] = (arr == cls).astype(int)
    return result, classes

sex_enc, sex_classes = one_hot(df['Sex'].values)
emb_enc, emb_classes = one_hot(df['Embarked'].fillna('S').values)

# ── Z-Score Normalizer ───────────────────────────────────────
def zscore(X):
    mu  = X.mean(axis=0)
    std = X.std(axis=0)
    std[std == 0] = 1  # prevent division by zero
    return (X - mu) / std, mu, std

# ── Min-Max Normalizer ───────────────────────────────────────
def minmax(X):
    mn  = X.min(axis=0)
    mx  = X.max(axis=0)
    rng = mx - mn
    rng[rng == 0] = 1
    return (X - mn) / rng, mn, mx

# ── Assemble Feature Matrix ──────────────────────────────────
num_features = df[['Age','Fare','FamilySize','Pclass']].values
X = np.hstack([num_features, sex_enc, emb_enc,
               df[['IsAlone']].values])
X_norm, mu, std = zscore(X.astype(float))
y = df['Survived'].values

# ── Correlation Matrix from Scratch ─────────────────────────
def correlation_matrix(X):
    X_c = X - X.mean(axis=0)
    std = X.std(axis=0)
    std[std == 0] = 1
    X_n = X_c / std
    return (X_n.T @ X_n) / len(X)

corr = correlation_matrix(X_norm)
print("Correlation matrix shape:", corr.shape)

Logistic Regression from Scratch

Sigmoid, binary cross-entropy loss, gradient descent with L2 regularization.

The math: We predict P(survive) = σ(Xw + b). Loss = −[y·log(ŷ) + (1−y)·log(1−ŷ)] + λ‖w‖². Gradients flow back through the sigmoid to update weights. No sklearn. Just NumPy and calculus.
logistic_regression.py 
class LogisticRegression:
    def __init__(self, lr=0.01, epochs=1000, lambda_=0.01):
        self.lr      = lr
        self.epochs  = epochs
        self.lambda_ = lambda_
        self.losses  = []

    def _sigmoid(self, z):
        return 1 / (1 + np.exp(-np.clip(z, -500, 500)))

    def _loss(self, y, y_hat):
        eps = 1e-15
        y_hat = np.clip(y_hat, eps, 1 - eps)
        bce  = -np.mean(y * np.log(y_hat) + (1-y) * np.log(1-y_hat))
        l2   = (self.lambda_ / (2 * len(y))) * np.sum(self.w**2)
        return bce + l2

    def fit(self, X, y):
        n, d      = X.shape
        self.w    = np.zeros(d)
        self.b    = 0.0
        self.losses = []

        for epoch in range(self.epochs):
            z     = X @ self.w + self.b
            y_hat = self._sigmoid(z)
            err   = y_hat - y

            # Gradients
            dw = (X.T @ err) / n + (self.lambda_ / n) * self.w
            db = np.mean(err)

            # Update
            self.w -= self.lr * dw
            self.b -= self.lr * db

            if epoch % 10 == 0:
                loss = self._loss(y, y_hat)
                self.losses.append(loss)

        return self

    def predict_proba(self, X):
        return self._sigmoid(X @ self.w + self.b)

    def predict(self, X, threshold=0.5):
        return (self.predict_proba(X) >= threshold).astype(int)

# ── Train ────────────────────────────────────────────────────
model = LogisticRegression(lr=0.1, epochs=2000, lambda_=0.01)
model.fit(X_norm, y)
print(f"Final loss: {model.losses[-1]:.4f}")

evaluation.py 
# ── Metrics from Scratch ─────────────────────────────────────
def confusion_matrix(y_true, y_pred):
    TP = np.sum((y_true == 1) & (y_pred == 1))
    TN = np.sum((y_true == 0) & (y_pred == 0))
    FP = np.sum((y_true == 0) & (y_pred == 1))
    FN = np.sum((y_true == 1) & (y_pred == 0))
    return np.array([[TN, FP], [FN, TP]])

def metrics(y_true, y_pred):
    cm        = confusion_matrix(y_true, y_pred)
    TN,FP,FN,TP = cm[0,0],cm[0,1],cm[1,0],cm[1,1]
    accuracy  = (TP + TN) / len(y_true)
    precision = TP / (TP + FP + 1e-10)
    recall    = TP / (TP + FN + 1e-10)
    f1        = 2 * precision * recall / (precision + recall + 1e-10)
    return {'accuracy':accuracy, 'precision':precision,
            'recall':recall, 'f1':f1, 'confusion':cm}

# ── K-Fold Cross Validation ──────────────────────────────────
def kfold_cv(X, y, k=5, lr=0.1, epochs=1000):
    n      = len(X)
    size   = n // k
    idx    = np.arange(n)
    np.random.shuffle(idx)
    scores = []

    for fold in range(k):
        val_idx   = idx[fold*size:(fold+1)*size]
        train_idx = np.concatenate([idx[:fold*size],
                                     idx[(fold+1)*size:]])
        X_tr, X_val = X[train_idx], X[val_idx]
        y_tr, y_val = y[train_idx], y[val_idx]

        m = LogisticRegression(lr=lr, epochs=epochs)
        m.fit(X_tr, y_tr)

        preds = m.predict(X_val)
        m_dict = metrics(y_val, preds)
        scores.append(m_dict['accuracy'])
        print(f"Fold {fold+1}: acc={m_dict['accuracy']:.3f}")

    print(f"Mean CV accuracy: {np.mean(scores):.3f} ± {np.std(scores):.3f}")
    return scores

scores = kfold_cv(X_norm, y, k=5)

def preprocess_passenger(p: dict, mu, std) -> np.ndarray:
    """Full pipeline: raw dict → normalized feature vector."""
    age        = p['Age']
    fare       = np.log1p(min(p['Fare'], 65.0))  # IQR cap
    family     = p['SibSp'] + p['Parch'] + 1
    is_alone   = 1 if family == 1 else 0
    sex_female = 1 if p['Sex'] == 'female' else 0
    sex_male   = 1 - sex_female
    emb_c = 1 if p['Embarked']=='C' else 0
    emb_q = 1 if p['Embarked']=='Q' else 0
    emb_s = 1 if p['Embarked']=='S' else 0

    raw = np.array([age, fare, family, p['Pclass'],
                    sex_female, sex_male,
                    emb_c, emb_q, emb_s, is_alone], dtype=float)
    return (raw - mu) / std

def predict_passenger(passenger_dict):
    x = preprocess_passenger(passenger_dict, mu, std)
    prob = model.predict_proba(x.reshape(1, -1))[0]
    verdict = "SURVIVED" if prob >= 0.5 else "DID NOT SURVIVE"
    print(f"Survival probability: {prob:.1%} → {verdict}")
    return prob
