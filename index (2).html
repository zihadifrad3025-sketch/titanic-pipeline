Phase 2 — NumPy Encoding & Normalization

+

Built a one-hot encoder, Z-score normalizer, and Min-Max normalizer from scratch using only NumPy — no pd.get_dummies or sklearn preprocessing.

def one_hot(arr):
    classes = np.unique(arr)
    result = np.zeros((len(arr), len(classes)), dtype=int)
    for i, cls in enumerate(classes):
        result[:, i] = (arr == cls).astype(int)
    return result, classes

def zscore(X, mu=None, std=None):
    if mu is None:
        mu = X.mean(axis=0)
        std = X.std(axis=0)
    std[std == 0] = 1
    return (X - mu) / std, mu, std

def _sigmoid(self, z):
    return 1 / (1 + np.exp(-np.clip(z, -500, 500)))

def _loss(self, y, y_hat):
    eps = 1e-15
    y_hat = np.clip(y_hat, eps, 1 - eps)
    bce = -np.mean(y * np.log(y_hat) + (1-y) * np.log(1-y_hat))
    l2  = (self.lambda_ / (2 * len(y))) * np.sum(self.w ** 2)
    return bce + l2

Phase 4 — Evaluation Without sklearn

+

All evaluation metrics were built from scratch — Accuracy, Precision, Recall, F1, and Confusion Matrix. Also implemented 5-fold cross validation manually using array slicing.

def compute_metrics(y_true, y_pred):
    TP = np.sum((y_true == 1) & (y_pred == 1))
    TN = np.sum((y_true == 0) & (y_pred == 0))
    FP = np.sum((y_true == 0) & (y_pred == 1))
    FN = np.sum((y_true == 1) & (y_pred == 0))
    accuracy  = (TP + TN) / len(y_true)
    precision = TP / (TP + FP + 1e-10)
    recall    = TP / (TP + FN + 1e-10)
    f1 = 2 * precision * recall / (precision + recall + 1e-10)
    return accuracy, precision, recall, f1
